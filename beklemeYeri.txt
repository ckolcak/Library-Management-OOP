void Member::addBook(int bookID){
    if(borrowedBookIDs.size() >= maxBorrowLimit){
        return;
    }
    if(hasBook(bookID)){
        return;
    }
    borrowedBookIDs.push_back(bookID);
}

void Member::subtractBook(int bookID){
    if(!hasBook(bookID)){
        return;
    }
    borrowedBookIDs.erase(std::remove(borrowedBookIDs.begin(), borrowedBookIDs.end(), bookID), borrowedBookIDs.end());//erase-remove idiomu (#include <algorithm> lazım)
}

bool Member::hasBook(int bookID) const{
    for(int i = 0; i < borrowedBookIDs.size(); i++){
        if(borrowedBookIDs[i] == bookID){
        return true;
        }
    }
        return false;
}
